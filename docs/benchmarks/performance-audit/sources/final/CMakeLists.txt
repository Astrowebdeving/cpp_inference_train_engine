cmake_minimum_required(VERSION 3.28)
project(mini_mla_completed VERSION 1.0 LANGUAGES CXX OBJCXX)

if(NOT APPLE)
    message(FATAL_ERROR "This engine requires macOS and Accelerate.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(MINI_MLA_ENABLE_MPS "Enable MPSGraph execution" ON)
option(MINI_MLA_ENABLE_ASAN "Enable AddressSanitizer" OFF)
option(MINI_MLA_ENABLE_UBSAN "Enable UndefinedBehaviorSanitizer" OFF)
option(MINI_MLA_BUILD_TESTS "Build tests" ON)
option(MINI_MLA_WARNINGS_AS_ERRORS "Treat warnings as errors" ON)
option(MINI_MLA_ENABLE_GEMMA "Build the quantized Gemma text runner with MLX C++" OFF)

add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(MINI_MLA_WARNINGS_AS_ERRORS)
    add_compile_options(-Werror)
endif()
if(MINI_MLA_ENABLE_ASAN)
    add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address)
endif()
if(MINI_MLA_ENABLE_UBSAN)
    add_compile_options(-fsanitize=undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=undefined)
endif()
find_library(ACCELERATE Accelerate REQUIRED)
find_library(FOUNDATION Foundation REQUIRED)
find_library(METAL Metal REQUIRED)
find_library(MPS MetalPerformanceShaders REQUIRED)
find_library(MPSGRAPH MetalPerformanceShadersGraph REQUIRED)
find_package(Threads REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/backends/flash_attention.metal)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/backends/flash_attention.metal" MINI_MLA_FLASH_SOURCE)
configure_file(src/backends/flash_attention_source.hpp.in generated/flash_attention_source.hpp @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/backends/adamw.metal)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/backends/adamw.metal" MINI_MLA_ADAMW_SOURCE)
configure_file(src/backends/adamw_source.hpp.in generated/adamw_source.hpp @ONLY)

add_library(mini_mla STATIC
    src/core/result.cpp src/core/dtype.cpp src/core/shape.cpp src/core/buffer.cpp
    src/core/tensor.cpp src/core/json.cpp src/core/io.cpp
    src/backends/cpu_backend.cpp src/backends/mps_backend.mm
    src/model/config.cpp src/model/ops.cpp src/model/model.cpp
    src/train/optimizer.cpp src/train/data.cpp src/train/checkpoint.cpp src/train/trainer.cpp
    src/infer/tokenizer.cpp src/infer/inference.cpp src/serve/server.cpp)
target_include_directories(mini_mla PUBLIC include)
target_include_directories(mini_mla PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(mini_mla PUBLIC MINI_MLA_ENABLE_MPS=$<BOOL:${MINI_MLA_ENABLE_MPS}> ACCELERATE_NEW_LAPACK)
target_compile_options(mini_mla PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
target_link_libraries(mini_mla PUBLIC ${ACCELERATE} ${FOUNDATION} ${METAL} ${MPS} ${MPSGRAPH} Threads::Threads)

foreach(app IN ITEMS info tokenize train generate inspect serve bench)
    add_executable(mini_mla_${app} apps/${app}_main.cpp)
    target_link_libraries(mini_mla_${app} PRIVATE mini_mla)
endforeach()
if(MINI_MLA_ENABLE_GEMMA)
    find_package(MLX CONFIG REQUIRED)
    get_target_property(mlx_links mlx INTERFACE_LINK_LIBRARIES)
    set(mlx_resolved_links "")
    foreach(dependency IN LISTS mlx_links)
        if(dependency MATCHES "\\.framework$" AND NOT EXISTS "${dependency}")
            get_filename_component(framework_name "${dependency}" NAME_WE)
            find_library(MINI_MLA_MLX_${framework_name} ${framework_name} REQUIRED)
            list(APPEND mlx_resolved_links "${MINI_MLA_MLX_${framework_name}}")
        else()
            list(APPEND mlx_resolved_links "${dependency}")
        endif()
    endforeach()
    set_property(TARGET mlx PROPERTY INTERFACE_LINK_LIBRARIES "${mlx_resolved_links}")
    add_executable(mini_mla_gemma apps/gemma_main.cpp src/infer/gemma.cpp)
    target_link_libraries(mini_mla_gemma PRIVATE mini_mla mlx)
endif()
if(MINI_MLA_BUILD_TESTS)
    enable_testing()
    foreach(test IN ITEMS tensor ops model files training mps)
        add_executable(test_${test} tests/test_${test}.cpp)
        target_link_libraries(test_${test} PRIVATE mini_mla)
        add_test(NAME ${test} COMMAND test_${test})
        set_tests_properties(${test} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 180)
    endforeach()
    set_tests_properties(mps PROPERTIES SKIP_RETURN_CODE 77)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME tools COMMAND ${Python3_EXECUTABLE} tests/test_tools.py "${CMAKE_BINARY_DIR}/bin")
        set_tests_properties(tools PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 180)
    endif()
endif()
